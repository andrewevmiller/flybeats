cmake_minimum_required(VERSION 3.20)
project(flybrain_voices LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FLYBRAIN_VOICES_SANITIZE "Build with address and undefined-behaviour sanitizers" OFF)
option(FLYBRAIN_VOICES_BUILD_TESTS "Build the tests and the render tool" ON)

add_library(flybrain_voices STATIC src/voices.cpp)
target_include_directories(flybrain_voices PUBLIC include)
if(NOT MSVC)
  target_compile_options(flybrain_voices PRIVATE -Wall -Wextra)
endif()
if(FLYBRAIN_VOICES_SANITIZE AND NOT MSVC)
  target_compile_options(flybrain_voices PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(flybrain_voices PUBLIC -fsanitize=address,undefined)
endif()

if(FLYBRAIN_VOICES_BUILD_TESTS)
  enable_testing()
  add_executable(voices_tests tests/voices_tests.cpp)
  target_link_libraries(voices_tests PRIVATE flybrain_voices)
  add_test(NAME voices_tests COMMAND voices_tests)
  add_executable(voices_alloc_test tests/voices_alloc_test.cpp)
  target_link_libraries(voices_alloc_test PRIVATE flybrain_voices)
  add_test(NAME voices_alloc_test COMMAND voices_alloc_test)
  add_executable(voices_render tools/voices_render.cpp)
  target_link_libraries(voices_render PRIVATE flybrain_voices)
endif()
